cmake_minimum_required(VERSION 3.22)
project(WhiskerMix VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
option(WM_WINDOWS_TEST "JUCE audio device instead of PipeWire/libusb (Windows and macOS builds, same UI)" OFF)
option(WM_BUILD_PLUGINS "Build the standalone effect plugins (VST3 + Standalone)" OFF)
option(WM_BUILD_APP "Build the JUCE + PipeWire app" OFF)

enable_testing()
foreach(t digiface_codec_test names_test mixer_test state_test engine_test reverb_test hub_test)
  add_executable(${t} tests/${t}.cpp)
  add_test(NAME ${t} COMMAND ${t})
endforeach()

if(WM_BUILD_APP)
  if(NOT WM_WINDOWS_TEST)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(PW REQUIRED libpipewire-0.3)
    pkg_check_modules(USB REQUIRED libusb-1.0)
  endif()
  add_subdirectory(third_party/JUCE)   # JUCE added as a vendored source (needed for offline Flatpak builds)
  juce_add_gui_app(WhiskerMix PRODUCT_NAME "WhiskerMix")
  target_sources(WhiskerMix PRIVATE src/app/Main.cpp)
  target_compile_definitions(WhiskerMix PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_APPLICATION_NAME_STRING="WhiskerMix" JUCE_APPLICATION_VERSION_STRING="0.1.0")
  target_link_libraries(WhiskerMix PRIVATE juce::juce_audio_devices juce::juce_gui_basics)
  if(WM_WINDOWS_TEST)
    target_compile_definitions(WhiskerMix PRIVATE WM_JUCE_AUDIO=1)
  else()
    target_sources(WhiskerMix PRIVATE src/pipewire/PipeWireMixer.cpp src/backends/digiface/LibusbTransport.cpp)
    target_include_directories(WhiskerMix PRIVATE ${PW_INCLUDE_DIRS} ${USB_INCLUDE_DIRS})
    target_link_libraries(WhiskerMix PRIVATE ${PW_LIBRARIES} ${USB_LIBRARIES})
  endif()
endif()

if(WM_BUILD_PLUGINS)
  if(NOT TARGET juce::juce_audio_processors)
    add_subdirectory(third_party/JUCE)
  endif()
  foreach(plug ShimmerDelay ShimmerReverb)
    if(plug STREQUAL "ShimmerDelay")
      set(src plugins/shimmer-delay/ShimmerDelayPlugin.cpp)
      set(code Shdl)
    else()
      set(src plugins/shimmer-reverb/ShimmerReverbPlugin.cpp)
      set(code Shrv)
    endif()
    juce_add_plugin(${plug}
      PRODUCT_NAME "${plug}"
      COMPANY_NAME "SpeakingColors"
      PLUGIN_MANUFACTURER_CODE Spco
      PLUGIN_CODE ${code}
      FORMATS VST3 Standalone
      IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE)
    target_sources(${plug} PRIVATE ${src})
    target_compile_definitions(${plug} PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
    target_link_libraries(${plug} PRIVATE juce::juce_audio_utils juce::juce_audio_processors juce::juce_recommended_config_flags)
  endforeach()
endif()
